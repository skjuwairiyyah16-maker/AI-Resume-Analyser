SQL Setup Commands (`database.sql`) MySQL 8.0 
CREATE DATABASE IF NOT EXISTS resume_analyzer_db;
USE resume_analyzer_db;

CREATE TABLE users (
    user_id INT AUTO_INCREMENT PRIMARY KEY,
    full_name VARCHAR(100) NOT NULL,
    email VARCHAR(120) UNIQUE NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE resumes (
    resume_id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT,
    filename VARCHAR(255) NOT NULL,
    raw_text LONGTEXT NOT NULL,
    uploaded_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(user_id) ON DELETE CASCADE
);

CREATE TABLE analyses (
    analysis_id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT,
    resume_id INT,
    target_role VARCHAR(100) NOT NULL,
    match_score INT NOT NULL,
    detected_skills TEXT,
    missing_skills TEXT,
    analyzed_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(user_id),
    FOREIGN KEY (resume_id) REFERENCES resumes(resume_id)
);

BCA Viva & Technical Interview Guide

Essential answers to expected examiner questions regarding algorithm mechanics and architecture.

Scoring Formula Mechanics

Final ATS Score = (TF-IDF Cosine Similarity × 70%) + (Section Completeness × 30%)

Where Cosine Similarity = (Vector A · Vector B) / (||Vector A|| × ||Vector B||)

Q1: Why use TF-IDF vectorization instead of deep learning (LLMs)?

"TF-IDF provides 100% mathematical explainability, runs locally with zero API cost, yields deterministic output without hallucination, and executes with sub-millisecond computational overhead."

Q2: How is password authentication secured?

"Passwords are never saved in plain text. We utilize salted SHA-256 / PBKDF2 hashing routines (`Werkzeug.security.generate_password_hash`) to ensure zero raw credential exposure."

Q3: How does text parsing process PDF content?

"The system reads byte stream arrays, tokenizes word elements using regex boundary matchers, converts tokens to lowercase, filters NLTK stop-words, and cross-references extracted tokens against a technical skill taxonomy."
